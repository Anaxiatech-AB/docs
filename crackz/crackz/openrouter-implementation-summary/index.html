    Architecture Highlights¶

    Provider Abstraction 

    class AIProvider(str, Enum):
    ANTHROPIC = "anthropic"
    OPENROUTER = "openrouter"

    Model Mapping 

    def _map_model_name(self, model: str, provider: str) -> str:
    """Maps model name to provider-specific format."""
    if provider == AIProvider.OPENROUTER:
        # Add provider prefix for OpenRouter
        if not model.startswith(("anthropic/", "openai/", "meta-llama/", "google/")):
            return f"anthropic/{model}"
    return model

    Provider Dispatch 

    async def _call_ai_api(self, message: str, context: list[dict[str, str]]) -> str:
    """Provider-agnostic API call dispatcher."""
    if self.provider == AIProvider.ANTHROPIC:
        return await self._call_anthropic_api(message, context)
    elif self.provider == AIProvider.OPENROUTER:
        return await self._call_openrouter_api(message, context)
    else:
        raise ValueError(f"Unsupported provider: {self.provider}")
