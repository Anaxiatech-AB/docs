    Model Naming & Registry Implementation Plan¶

    For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

    Goal: Replace raw checkpoint filenames in the GUI with user-friendly model names backed by a per-project SQLite registry.

    Architecture: A new ModelRecord Pydantic model and ModelRegistry SQLite wrapper live in src/crackz/ai/model/registry.py. Training auto-registers a model on completion. The model selector GUI queries the registry and shows model names with a detail panel; raw checkpoint names are kept as a fallback.

    Tech Stack: Python sqlite3 (stdlib), Pydantic v2, PySide6, Typer

    """Per-project SQLite model registry."""

from __future__ import annotations

import sqlite3
from datetime import datetime
from pathlib import Path

from pydantic import BaseModel

_SCHEMA_VERSION = 1

_CREATE_TABLE = """
CREATE TABLE IF NOT EXISTS models (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    name            TEXT    NOT NULL,
    checkpoint_name TEXT    NOT NULL,
    encoder         TEXT    NOT NULL,
    encoder_weights TEXT,
    in_channels     INTEGER NOT NULL,
    num_classes     INTEGER NOT NULL,
    threshold       REAL    NOT NULL,
    loss_name       TEXT    NOT NULL,
    dice_weight     REAL    NOT NULL,
    focal_weight    REAL    NOT NULL,
    created_at      TEXT    NOT NULL,
    notes           TEXT
)
"""


class ModelRecord(BaseModel):
    id: int | None = None
    name: str
    checkpoint_name: str
    encoder: str
    encoder_weights: str | None
    in_channels: int
    num_classes: int
    threshold: float
    loss_name: str
    dice_weight: float
    focal_weight: float
    created_at: datetime
    notes: str | None = None


class ModelRegistry:
    def __init__(self, db_path: Path) -> None:
        self._db_path = db_path
        self._init_db()

    # ------------------------------------------------------------------
    # Public helpers
    # ------------------------------------------------------------------

    @staticmethod
    def default_name(encoder: str, created_at: datetime) -> str:
        """Return the auto-generated display name for a model."""
        return f"{encoder}-{created_at:%Y-%m-%d}"

    # ------------------------------------------------------------------
    # CRUD
    # ------------------------------------------------------------------

    def register(self, record: ModelRecord) -> ModelRecord:
        """Insert a new model record; returns the record with id populated."""
        with self._connect() as con:
            cur = con.execute(
                """
                INSERT INTO models
                    (name, checkpoint_name, encoder, encoder_weights,
                     in_channels, num_classes, threshold, loss_name,
                     dice_weight, focal_weight, created_at, notes)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    record.name,
                    record.checkpoint_name,
                    record.encoder,
                    record.encoder_weights,
                    record.in_channels,
                    record.num_classes,
                    record.threshold,
                    record.loss_name,
                    record.dice_weight,
                    record.focal_weight,
                    record.created_at.isoformat(),
                    record.notes,
                ),
            )
            return record.model_copy(update={"id": cur.lastrowid})

    def get(self, id: int) -> ModelRecord | None:
        """Fetch a single model record by id."""
        with self._connect() as con:
            row = con.execute("SELECT * FROM models WHERE id = ?", (id,)).fetchone()
        return _row_to_record(row) if row else None

    def list_all(self) -> list[ModelRecord]:
        """Return all model records ordered by created_at descending."""
        with self._connect() as con:
            rows = con.execute(
                "SELECT * FROM models ORDER BY created_at DESC"
            ).fetchall()
        return [_row_to_record(r) for r in rows]

    def rename(self, id: int, name: str) -> None:
        """Update the display name of a model."""
        with self._connect() as con:
            con.execute("UPDATE models SET name = ? WHERE id = ?", (name, id))

    def update_notes(self, id: int, notes: str) -> None:
        """Update the free-text notes of a model."""
        with self._connect() as con:
            con.execute("UPDATE models SET notes = ? WHERE id = ?", (notes, id))

    def delete(self, id: int) -> None:
        """Delete a model record."""
        with self._connect() as con:
            con.execute("DELETE FROM models WHERE id = ?", (id,))

    def get_by_checkpoint(self, checkpoint_name: str) -> ModelRecord | None:
        """Return the most recently registered record for a checkpoint name."""
        with self._connect() as con:
            row = con.execute(
                "SELECT * FROM models WHERE checkpoint_name = ? ORDER BY created_at DESC LIMIT 1",
                (checkpoint_name,),
            ).fetchone()
        return _row_to_record(row) if row else None

    # ------------------------------------------------------------------
    # Internal
    # ------------------------------------------------------------------

    def _connect(self) -> sqlite3.Connection:
        return sqlite3.connect(self._db_path)

    def _init_db(self) -> None:
        with self._connect() as con:
            con.execute(_CREATE_TABLE)
            con.execute(f"PRAGMA user_version = {_SCHEMA_VERSION}")


def _row_to_record(row: tuple) -> ModelRecord:
    # Column order matches SELECT * from schema definition
    (id_, name, ckpt, enc, enc_w, in_ch, n_cls, thr, loss, dw, fw, ts, notes) = row
    return ModelRecord(
        id=id_,
        name=name,
        checkpoint_name=ckpt,
        encoder=enc,
        encoder_weights=enc_w,
        in_channels=in_ch,
        num_classes=n_cls,
        threshold=thr,
        loss_name=loss,
        dice_weight=dw,
        focal_weight=fw,
        created_at=datetime.fromisoformat(ts),
        notes=notes,
    )


__all__ = ["ModelRecord", "ModelRegistry"]

    """Model selector dialog — shows registry model names with metadata detail panel."""

from __future__ import annotations

from typing import TYPE_CHECKING

from PySide6.QtCore import Qt
from PySide6.QtWidgets import (
    QDialog,
    QDialogButtonBox,
    QFormLayout,
    QGroupBox,
    QHBoxLayout,
    QLabel,
    QListWidget,
    QListWidgetItem,
    QMessageBox,
    QPushButton,
    QSizePolicy,
    QVBoxLayout,
    QWidget,
)

from crackz.ai.model.registry import ModelRecord, ModelRegistry

if TYPE_CHECKING:
    from pathlib import Path

    from crackz import Project


def open_model_selector_dialog(parent: QWidget, project: Project) -> str | None | bool:
    """Open model selector dialog.

    Returns: checkpoint name string, None for auto-select, False if cancelled.
    """
    dlg = _ModelSelectorDialog(parent, project)
    if dlg.exec() != QDialog.DialogCode.Accepted:
        return False
    return dlg.selected_checkpoint


class _ModelSelectorDialog(QDialog):
    def __init__(self, parent: QWidget, project: Project) -> None:
        super().__init__(parent)
        self._project = project
        self._registry = ModelRegistry(project.registry_db_path)
        self._records: list[ModelRecord] = self._registry.list_all()
        self._registry_mode = bool(self._records)
        self.selected_checkpoint: str | None = None
        self.setWindowTitle("Select Model for Detection")
        self.setMinimumWidth(560)
        self.setMinimumHeight(420)
        self._build_ui()

    def _build_ui(self) -> None:
        layout = QVBoxLayout(self)

        if self._records:
            self._build_registry_ui(layout)
        else:
            self._build_fallback_ui(layout)

        buttons = QDialogButtonBox(
            QDialogButtonBox.StandardButton.Ok | QDialogButtonBox.StandardButton.Cancel
        )
        buttons.accepted.connect(self._on_ok)
        buttons.rejected.connect(self.reject)
        layout.addWidget(buttons)

    # ------------------------------------------------------------------
    # Registry mode (models exist in DB)
    # ------------------------------------------------------------------

    def _build_registry_ui(self, layout: QVBoxLayout) -> None:
        layout.addWidget(QLabel(
            "Select a model to use for detection.\n"
            "Click a model to see its details."
        ))

        # Auto-select option at top
        self._auto_label = QLabel("◉  Auto-select (best available checkpoint)")
        self._auto_label.setProperty("selected", True)
        self._auto_label.mousePressEvent = lambda _: self._select_auto()
        layout.addWidget(self._auto_label)
        self._auto_selected = True

        # List of named models
        self._list = QListWidget()
        self._list.setSizePolicy(QSizePolicy.Policy.Expanding, QSizePolicy.Policy.Expanding)
        for rec in self._records:
            item = QListWidgetItem(rec.name)
            item.setData(Qt.ItemDataRole.UserRole, rec)
            self._list.addItem(item)
        self._list.currentItemChanged.connect(self._on_item_changed)
        layout.addWidget(self._list)

        # Detail panel
        detail_box = QGroupBox("Model Details")
        self._detail_layout = QFormLayout(detail_box)
        self._detail_fields: dict[str, QLabel] = {}
        for field in ("Encoder", "Weights", "Threshold", "Loss", "In channels", "Classes", "Created", "Notes"):
            lbl = QLabel("—")
            lbl.setWordWrap(True)
            self._detail_layout.addRow(field + ":", lbl)
            self._detail_fields[field] = lbl
        layout.addWidget(detail_box)

        # Rename button
        rename_row = QHBoxLayout()
        self._rename_btn = QPushButton("Rename…")
        self._rename_btn.setEnabled(False)
        self._rename_btn.clicked.connect(self._on_rename)
        rename_row.addStretch()
        rename_row.addWidget(self._rename_btn)
        layout.addLayout(rename_row)

    def _select_auto(self) -> None:
        self._auto_selected = True
        self._list.clearSelection()
        self._rename_btn.setEnabled(False)
        self._clear_detail_panel()

    def _on_item_changed(self, current: QListWidgetItem | None, _prev: QListWidgetItem | None) -> None:
        if current is None:
            return
        self._auto_selected = False
        rec: ModelRecord = current.data(Qt.ItemDataRole.UserRole)
        self._populate_detail_panel(rec)
        self._rename_btn.setEnabled(True)

    def _populate_detail_panel(self, rec: ModelRecord) -> None:
        self._detail_fields["Encoder"].setText(rec.encoder)
        self._detail_fields["Weights"].setText(rec.encoder_weights or "—")
        self._detail_fields["Threshold"].setText(str(rec.threshold))
        self._detail_fields["Loss"].setText(
            f"{rec.loss_name} (dice={rec.dice_weight}, focal={rec.focal_weight})"
        )
        self._detail_fields["In channels"].setText(str(rec.in_channels))
        self._detail_fields["Classes"].setText(str(rec.num_classes))
        self._detail_fields["Created"].setText(rec.created_at.strftime("%Y-%m-%d %H:%M"))
        self._detail_fields["Notes"].setText(rec.notes or "—")

    def _clear_detail_panel(self) -> None:
        for lbl in self._detail_fields.values():
            lbl.setText("—")

    def _on_rename(self) -> None:
        item = self._list.currentItem()
        if item is None:
            return
        rec: ModelRecord = item.data(Qt.ItemDataRole.UserRole)
        from PySide6.QtWidgets import QInputDialog
        new_name, ok = QInputDialog.getText(
            self, "Rename Model", "New name:", text=rec.name
        )
        if ok and new_name.strip():
            new_name = new_name.strip()
            self._registry.rename(rec.id, new_name)
            updated = rec.model_copy(update={"name": new_name})
            item.setText(new_name)
            item.setData(Qt.ItemDataRole.UserRole, updated)

    def _on_ok(self) -> None:
        if self._registry_mode:
            if self._auto_selected:
                self.selected_checkpoint = None
            else:
                item = self._list.currentItem()
                if item is None:
                    self.selected_checkpoint = None
                else:
                    rec: ModelRecord = item.data(Qt.ItemDataRole.UserRole)
                    self.selected_checkpoint = rec.checkpoint_name
        else:
            checked = self._group.checkedButton()
            if checked:
                self.selected_checkpoint = checked.property("ckpt")
        self.accept()

    # ------------------------------------------------------------------
    # Fallback mode (no registry entries — legacy project)
    # ------------------------------------------------------------------

    def _build_fallback_ui(self, layout: QVBoxLayout) -> None:
        from PySide6.QtWidgets import QButtonGroup, QRadioButton, QScrollArea

        layout.addWidget(QLabel(
            "No named models found. Showing raw checkpoints.\n"
            "Train a model to enable the model registry."
        ))

        scroll = QScrollArea()
        scroll.setWidgetResizable(True)
        scroll.setMinimumHeight(200)
        inner = QWidget()
        il = QVBoxLayout(inner)
        self._group = QButtonGroup(self)

        auto_rb = QRadioButton("Auto-select (best > last > final)")
        auto_rb.setChecked(True)
        auto_rb.setProperty("ckpt", None)
        self._group.addButton(auto_rb)
        il.addWidget(auto_rb)

        checkpoints = _get_available_checkpoints(self._project.models_path)
        for name, path in sorted(checkpoints.items()):
            rb = QRadioButton(_format_checkpoint_label(name, path))
            rb.setProperty("ckpt", name)
            self._group.addButton(rb)
            il.addWidget(rb)

            il.addWidget(QLabel("No checkpoints found in project."))

        il.addStretch()
        scroll.setWidget(inner)
        layout.addWidget(scroll)
        self._group.buttonClicked.connect(
            lambda btn: setattr(self, "selected_checkpoint", btn.property("ckpt"))
        )

# Helpers (used by fallback mode only)

def _get_available_checkpoints(checkpoints_dir: Path) -> dict[str, Path]:
    if not checkpoints_dir.exists():
        return {}
    checkpoints = {}
    for name in ("best", "last", "final"):
        p = checkpoints_dir / f"{name}.ckpt"
        if p.exists():
            checkpoints[name] = p
    for p in checkpoints_dir.glob("epoch-*.ckpt"):
        checkpoints[p.stem] = p
    return checkpoints


def _format_checkpoint_label(name: str, path: Path) -> str:
    try:
        size_str = f"{path.stat().st_size / (1024 * 1024):.1f} MB"
    except OSError:
        size_str = "unknown size"

    if name == "best":
        desc = "Highest validation IoU"
    elif name == "last":
        desc = "Most recent training state"
    elif name == "final":
        desc = "Final training checkpoint"
    elif name.startswith("epoch-"):
        iou = name.replace("epoch-", "")
        desc = f"Epoch checkpoint (IoU: {iou})"
    else:
        desc = "Custom checkpoint"

    return f"{name}  —  {desc}  ({size_str})"
