    Gallery Sub-Views Implementation Plan¶

    For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

    Goal: Add a Summary | Gallery toolbar toggle to all four detection/training views, where Gallery shows a paginated thumbnail grid with metadata badges.

    Architecture: Each view wraps its existing content in a QStackedWidget (page 0 = summary, page 1 = gallery). Gallery widgets are constructed lazily on first switch. Two new reusable gallery widgets: DetectionGalleryWidget (badged thumbnails from detection visualizations) and DatasetGalleryWidget (project split images with TRAIN/VAL/TEST badges). The ViewToolbar is already capable — views call set_actions() with two toggle buttons.

    Tech Stack: PySide6 6.7+, existing ImageGalleryView/_ThumbCell from image_gallery_view.py, DetectionResult from crackz.ai.detection.results, Catppuccin Mocha theme tokens from crackz.gui.style.theme.

"""Detection gallery widget — paginated thumbnail grid with severity/confidence badges."""

from __future__ import annotations

import time
from pathlib import Path
from typing import TYPE_CHECKING

from PySide6.QtCore import Qt
from PySide6.QtWidgets import (
    QGridLayout,
    QHBoxLayout,
    QLabel,
    QPushButton,
    QScrollArea,
    QSizePolicy,
    QVBoxLayout,
    QWidget,
)

from crackz.gui.qt.views.image_gallery_view import _ThumbCell
from crackz.gui.style import theme

if TYPE_CHECKING:
    from crackz import Project
    from crackz.ai.detection.results import DetectionResult
    from crackz.gui.qt.app_state import QtAppState
    from crackz.gui.qt.main_window import MainWindow

_THUMB = (150, 150)
_PAGE = 80
_COLS = 5
_THROTTLE_MS = 250


class BadgedThumbCell(_ThumbCell):
    """Thumbnail cell with severity badge and confidence value overlaid."""

    def __init__(
        self,
        path: Path,
        app_state: QtAppState,
        size: tuple[int, int],
        result: DetectionResult,
    ) -> None:
        super().__init__(path, app_state, size)

        is_crack = result.has_crack is True or bool(
            result.crack_severity and result.crack_severity != "No Crack"
        )
        is_suspected = result.crack_severity == "Suspected Crack"

        if is_crack and not is_suspected:
            badge_text = "CRACK"
            badge_color = theme.RED
        elif is_suspected:
            badge_text = "~"
            badge_color = theme.YELLOW
        else:
            badge_text = "OK"
            badge_color = theme.GREEN

        conf_str = f"{result.confidence:.2f}" if result.confidence is not None else ""

        # Badge label overlaid at bottom of image area
        self._badge_lbl = QLabel(badge_text)
        self._badge_lbl.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self._badge_lbl.setStyleSheet(
            f"background-color: {badge_color}; color: {theme.BASE};"
            f" font-size: 7pt; font-weight: bold; border-radius: 3px; padding: 1px 4px;"
        )

        self._conf_lbl = QLabel(conf_str)
        self._conf_lbl.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self._conf_lbl.setStyleSheet(
            f"color: {theme.SUBTEXT1}; font-size: 7pt; background: transparent;"
        )

        badge_row = QWidget()
        badge_row.setStyleSheet("background: transparent;")
        bl = QHBoxLayout(badge_row)
        bl.setContentsMargins(0, 0, 0, 0)
        bl.setSpacing(4)
        bl.addStretch()
        bl.addWidget(self._badge_lbl)
        bl.addWidget(self._conf_lbl)
        bl.addStretch()

        self.layout().addWidget(badge_row)  # type: ignore[union-attr]
        self.setFixedSize(size[0] + 10, size[1] + 50)


class DetectionGalleryWidget(QWidget):
    """Paginated gallery of detection result thumbnails with severity badges."""

    def __init__(
        self,
        app_state: QtAppState,
        window: MainWindow,
        project: Project,
        results: list[DetectionResult],
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)
        self._app_state = app_state
        self._window = window
        self._project = project
        self._results: list[DetectionResult] = list(results)
        self._page = 0
        self._last_update_ms: float = 0.0
        self._pending_refresh = False
        self._build_ui()

    def add_result(self, result: DetectionResult) -> None:
        """Append a live result (throttled to 4 fps)."""
        self._results.append(result)
        now = time.time() * 1000
        if self._pending_refresh:
            return
        if now - self._last_update_ms >= _THROTTLE_MS:
            self._render_page()
            self._last_update_ms = now
        else:
            self._pending_refresh = True
            from PySide6.QtCore import QTimer
            remaining = int(_THROTTLE_MS - (now - self._last_update_ms))
            QTimer.singleShot(remaining, self, self._throttled_render)

    def _throttled_render(self) -> None:
        self._pending_refresh = False
        self._last_update_ms = time.time() * 1000
        self._render_page()

    def _build_ui(self) -> None:
        outer = QVBoxLayout(self)
        outer.setContentsMargins(0, 0, 0, 0)
        outer.setSpacing(0)

        self._scroll = QScrollArea()
        self._scroll.setWidgetResizable(True)
        outer.addWidget(self._scroll, 1)

        self._page_bar = QWidget()
        pl = QHBoxLayout(self._page_bar)
        pl.setContentsMargins(8, 4, 8, 4)
        self._prev_btn = QPushButton("◀ Prev")
        self._prev_btn.setFixedWidth(80)
        self._prev_btn.clicked.connect(self._prev_page)
        pl.addWidget(self._prev_btn)
        self._page_lbl = QLabel()
        self._page_lbl.setAlignment(Qt.AlignmentFlag.AlignCenter)
        pl.addWidget(self._page_lbl, 1)
        self._next_btn = QPushButton("Next ▶")
        self._next_btn.setFixedWidth(80)
        self._next_btn.clicked.connect(self._next_page)
        pl.addWidget(self._next_btn)
        outer.addWidget(self._page_bar)

        self._render_page()

    def _render_page(self) -> None:
        results = self._results
        total = len(results)
        self._page_bar.setVisible(total > _PAGE)

        if not results:
            empty = QLabel("No detection results yet.")
            empty.setAlignment(Qt.AlignmentFlag.AlignCenter)
            empty.setStyleSheet(f"color: {theme.SUBTEXT1}; font-size: 12pt;")
            self._scroll.setWidget(empty)
            return

        total_pages = max(1, (total + _PAGE - 1) // _PAGE)
        self._page = min(self._page, total_pages - 1)
        self._page_lbl.setText(f"Page {self._page + 1} of {total_pages}")
        self._prev_btn.setEnabled(self._page > 0)
        self._next_btn.setEnabled((self._page + 1) * _PAGE < total)

        start = self._page * _PAGE
        page_results = results[start : start + _PAGE]

        container = QWidget()
        grid = QGridLayout(container)
        grid.setContentsMargins(8, 8, 8, 8)
        grid.setSpacing(6)

        vis_dir = self._project.detections_visualizations_path

        for idx, result in enumerate(page_results):
            img_path: Path | None = None
            if result.image and vis_dir:
                candidate = Path(vis_dir) / result.image
                if candidate.exists():
                    img_path = candidate
            if img_path is None and result.source_image:
                fallback = Path(self._project.paths.image_path) / result.source_image
                if fallback.exists():
                    img_path = fallback

            if img_path is None:
                img_path = Path(result.image or "unknown.png")

            cell = BadgedThumbCell(img_path, self._app_state, _THUMB, result)
            cell.clicked.connect(self._on_thumb_clicked)
            grid.addWidget(cell, idx // _COLS, idx % _COLS)

        remainder = len(page_results) % _COLS
        if remainder:
            for col in range(remainder, _COLS):
                spacer = QWidget()
                spacer.setSizePolicy(QSizePolicy.Policy.Expanding, QSizePolicy.Policy.Preferred)
                grid.addWidget(spacer, (len(page_results) - 1) // _COLS, col)

        self._scroll.setWidget(container)

    def _prev_page(self) -> None:
        if self._page > 0:
            self._page -= 1
            self._render_page()

    def _next_page(self) -> None:
        if (self._page + 1) * _PAGE < len(self._results):
            self._page += 1
            self._render_page()

    def _on_thumb_clicked(self, path: Path) -> None:
        from crackz.gui.qt.views.file_preview_view import FilePreviewView
        self._window.show_view(FilePreviewView(self._app_state, self._window, path))
        self._window.set_status(str(path))

_IMAGE_EXT = {".png", ".jpg", ".jpeg", ".bmp", ".tif", ".tiff", ".webp"}
_THUMB = (150, 150)
_PAGE = 80
_COLS = 5

_SPLIT_COLORS = {

class SplitThumbCell(_ThumbCell):
    """Thumbnail cell with a TRAIN / VAL / TEST split badge."""

        row = QWidget()

        rl = QHBoxLayout(row)
        rl.setContentsMargins(0, 0, 0, 0)
        rl.addStretch()
        rl.addWidget(badge)
        rl.addStretch()

        self.layout().addWidget(row)  # type: ignore[union-attr]
        self.setFixedSize(size[0] + 10, size[1] + 50)


class DatasetGalleryWidget(QWidget):
    """Paginated gallery of project dataset images with split badges."""

    def __init__(
        self,
        app_state: QtAppState,
        window: MainWindow,
        project: Project,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)
        self._app_state = app_state
        self._window = window
        self._project = project
        self._page = 0
        self._images: list[tuple[Path, str]] = self._collect_images()
        self._build_ui()

    def _collect_images(self) -> list[tuple[Path, str]]:
        images: list[tuple[Path, str]] = []
        for split_name, path_attr in (
            ("TRAIN", "train_path"),
            ("VAL", "val_path"),
            ("TEST", "test_path"),
        ):
            split_dir = getattr(self._project, path_attr, None)
            if split_dir and Path(split_dir).exists():
                for f in sorted(Path(split_dir).iterdir()):
                    if f.is_file() and f.suffix.lower() in _IMAGE_EXT:
                        images.append((f, split_name))
        return images

    def _build_ui(self) -> None:
        outer = QVBoxLayout(self)
        outer.setContentsMargins(0, 0, 0, 0)
        outer.setSpacing(0)

        self._scroll = QScrollArea()
        self._scroll.setWidgetResizable(True)
        outer.addWidget(self._scroll, 1)

        self._page_bar = QWidget()
        pl = QHBoxLayout(self._page_bar)
        pl.setContentsMargins(8, 4, 8, 4)
        self._prev_btn = QPushButton("◀ Prev")
        self._prev_btn.setFixedWidth(80)
        self._prev_btn.clicked.connect(self._prev_page)
        pl.addWidget(self._prev_btn)
        self._page_lbl = QLabel()
        self._page_lbl.setAlignment(Qt.AlignmentFlag.AlignCenter)
        pl.addWidget(self._page_lbl, 1)
        self._next_btn = QPushButton("Next ▶")
        self._next_btn.setFixedWidth(80)
        self._next_btn.clicked.connect(self._next_page)
        pl.addWidget(self._next_btn)
        outer.addWidget(self._page_bar)

        self._render_page()

    def _render_page(self) -> None:
        total = len(self._images)
        self._page_bar.setVisible(total > _PAGE)

        if not total:
            empty = QLabel("No dataset images found.\nImport images to populate this view.")
            empty.setAlignment(Qt.AlignmentFlag.AlignCenter)
            empty.setStyleSheet(f"color: {theme.SUBTEXT1}; font-size: 12pt;")
            self._scroll.setWidget(empty)
            return

        total_pages = max(1, (total + _PAGE - 1) // _PAGE)
        self._page = min(self._page, total_pages - 1)
        self._page_lbl.setText(f"Page {self._page + 1} of {total_pages}")
        self._prev_btn.setEnabled(self._page > 0)
        self._next_btn.setEnabled((self._page + 1) * _PAGE < total)

        start = self._page * _PAGE
        page_items = self._images[start : start + _PAGE]

        container = QWidget()
        grid = QGridLayout(container)
        grid.setContentsMargins(8, 8, 8, 8)
        grid.setSpacing(6)

        for idx, (img_path, split) in enumerate(page_items):
            cell = SplitThumbCell(img_path, self._app_state, _THUMB, split)
            cell.clicked.connect(self._on_thumb_clicked)
            grid.addWidget(cell, idx // _COLS, idx % _COLS)

        remainder = len(page_items) % _COLS
        if remainder:
            for col in range(remainder, _COLS):
                spacer = QWidget()
                spacer.setSizePolicy(QSizePolicy.Policy.Expanding, QSizePolicy.Policy.Preferred)
                grid.addWidget(spacer, (len(page_items) - 1) // _COLS, col)

        self._scroll.setWidget(container)

    def _prev_page(self) -> None:
        if self._page > 0:
            self._page -= 1
            self._render_page()

    def _next_page(self) -> None:
        if (self._page + 1) * _PAGE < len(self._images):
            self._page += 1
            self._render_page()

        from crackz.gui.qt.views.file_preview_view import FilePreviewView
