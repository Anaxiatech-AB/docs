    Qt Summary Views Redesign Implementation Plan¶

    For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

    Goal: Rewrite TrainingResultsView and DetectionResultsView in src/crackz/qt/views/ to use Catppuccin Mocha tokens from crackz.gui.style.theme, fix data parsing bugs, and surface additional metrics (best epoch, mean IoU, avg crack area).

    Architecture: Both views are BaseView subclasses in src/crackz/qt/views/. They are rewritten in-place — no file moves.     Background¶

    The current view: - Has a hard-coded #1F6AA5 blue header bar (not Catppuccin) - Uses #F57C00, #1976D2, #7B1FA2 for stat cards (Material Design, not Catppuccin) - Calls _parse_metrics() which ignores the epochs key and total_epochs field that the training callback writes specifically for GUI display - Has no "best epoch" concept

    The JSON written by MetricsLoggerCallback._save() has this shape: 

    {
  "epochs": [{"epoch": 0, "train_loss": 0.8, "val_loss": 0.7, ...}, ...],
  "total_epochs": 10,
  "train": [...],
  "val": [...],
  "test": [...],
  "updated_at": "2026-04-16T10:30:00Z"
}

    The epochs list contains merged train+val data per epoch, designed for GUI display. partial: true entries are mid-epoch snapshots and must be excluded from the summary.

from __future__ import annotations

import json
from typing import TYPE_CHECKING

from PySide6.QtCore import Qt
from PySide6.QtGui import QFont
from PySide6.QtWidgets import (
    QFrame,
    QGridLayout,
    QHBoxLayout,

from crackz.gui.style import theme
from crackz.gui.qt.views.base_view import BaseView

if TYPE_CHECKING:
    from crackz import Project
    from crackz.gui.qt.app_state import QtAppState
    from crackz.gui.qt.main_window import MainWindow

_GOOD = 0.7
_WARN = 0.5
_LOSS_WARN = 0.5


class TrainingResultsView(BaseView):
    """Scrollable training results view — reads training_metrics.json."""

    def __init__(
        self,
        app_state: QtAppState,
        window: MainWindow,
        project: Project,
        parent: QWidget | None = None,
    ) -> None:
        self._project = project
        super().__init__(app_state, window, parent)

    def _build_ui(self) -> None:
        outer = QVBoxLayout(self)
        outer.setContentsMargins(0, 0, 0, 0)
        outer.setSpacing(0)

        # Title row — CRUST background, no coloured header bar
        title_row = QFrame()
        title_row.setStyleSheet(f"background-color: {theme.CRUST};")
        tl = QVBoxLayout(title_row)
        tl.setContentsMargins(16, 12, 16, 12)
        tl.setSpacing(2)
        title_lbl = QLabel(f"Training Results — {self._project.project.name}")
        tf = QFont()
        tf.setPointSize(14)
        tf.setBold(True)
        title_lbl.setFont(tf)
        title_lbl.setStyleSheet(f"color: {theme.TEXT}; background: transparent;")
        tl.addWidget(title_lbl)
        outer.addWidget(title_row)

        # Scroll area
        scroll = QScrollArea()
        scroll.setWidgetResizable(True)
        scroll.setFrameShape(QFrame.Shape.NoFrame)
        scroll.setStyleSheet(f"QScrollArea {{ background: {theme.MANTLE}; }}")
        outer.addWidget(scroll, 1)

        content = QWidget()
        content.setStyleSheet(f"background: {theme.MANTLE};")
        layout = QVBoxLayout(content)
        layout.setContentsMargins(16, 16, 16, 16)
        layout.setSpacing(12)
        scroll.setWidget(content)

        metrics_path = self._project.training_path / "training_metrics.json"
        if not metrics_path.exists():
            msg = QLabel("No training metrics available.\nTrain a model to see results here.")
            msg.setAlignment(Qt.AlignmentFlag.AlignCenter)
            msg.setStyleSheet(
                f"color: {theme.SUBTEXT1}; font-size: 13pt; background: transparent;"
            )
            layout.addWidget(msg)
            layout.addStretch()
            return

        try:
            with metrics_path.open(encoding="utf-8") as f:
                raw = json.load(f)
        except Exception:
            err = QLabel("Could not load training metrics.")
            err.setAlignment(Qt.AlignmentFlag.AlignCenter)
            err.setStyleSheet(f"color: {theme.RED}; font-size: 13pt; background: transparent;")
            layout.addWidget(err)
            layout.addStretch()
            return

        latest, completed, total_epochs, updated_at = _parse_metrics(raw)
        if not latest:
            msg = QLabel("Training metrics file is empty.")
            msg.setAlignment(Qt.AlignmentFlag.AlignCenter)
            msg.setStyleSheet(
                f"color: {theme.SUBTEXT1}; font-size: 13pt; background: transparent;"
            )
            layout.addWidget(msg)
            layout.addStretch()
            return

        # Last trained subtitle
        if updated_at:
            subtitle = QLabel(f"Last trained: {_fmt_timestamp(updated_at)}")
            subtitle.setStyleSheet(
                f"color: {theme.SUBTEXT1}; font-size: 9pt; background: transparent;"
            )
            layout.addWidget(subtitle)

        # Epoch stat cards
        layout.addWidget(_section_label("Epoch Statistics"))
        layout.addWidget(_epoch_stats_row(completed, total_epochs, _best_epoch(raw)))

        # Phase metric grids — from last completed epoch
        train_m = {k: v for k, v in latest.items() if k.startswith("train_")}
        val_m = {k: v for k, v in latest.items() if k.startswith("val_")}
        test_m = {k: v for k, v in latest.items() if k.startswith("test_")}

        if train_m:
            layout.addWidget(_section_label("Training Metrics"))
            layout.addWidget(_metric_grid(train_m))
        if val_m:
            layout.addWidget(_section_label("Validation Metrics"))
            layout.addWidget(_metric_grid(val_m))
        if test_m:
            layout.addWidget(_section_label("Test Metrics"))
            layout.addWidget(_metric_grid(test_m))

        # Warnings
        warnings_path = self._project.training_path / "training_warnings.json"
        if warnings_path.exists():
            try:
                with warnings_path.open(encoding="utf-8") as f:
                    warnings_data = json.load(f)
                if warnings_data:
                    layout.addWidget(_warnings_widget(warnings_data))
            except Exception:
                pass

        layout.addStretch()


# ── Data helpers ──────────────────────────────────────────────────────────────


def _parse_metrics(data: dict | list) -> tuple[dict | None, int, int, str | None]:
    """Parse training_metrics.json.

    Returns (latest_epoch, completed_epochs, total_epochs, updated_at).

    Prefers the 'epochs' key (merged train+val, designed for GUI display).
    Partial entries (mid-epoch snapshots written every 10 batches) are excluded.
    Falls back to 'train'/'val'/'test' lists for files without the 'epochs' key.
    """
    if isinstance(data, list) and data:
        non_partial = [e for e in data if not e.get("partial")]
        latest = non_partial[-1] if non_partial else data[-1]
        n = len(non_partial)
        return latest, n, n, None

    if not isinstance(data, dict):
        return None, 0, 0, None

    updated_at = data.get("updated_at")
    total_epochs = data.get("total_epochs") or 0

    epochs = data.get("epochs", [])
    if epochs:
        non_partial = [e for e in epochs if not e.get("partial")]
        if non_partial:
            completed = len(non_partial)
            return non_partial[-1], completed, total_epochs or completed, updated_at

    # Fallback for older files without 'epochs' key
    for key in ("train", "val", "test"):
        subset = data.get(key, [])
        if subset:
            completed = len(subset)
            return subset[-1], completed, total_epochs or completed, updated_at

    return None, 0, 0, updated_at


def _best_epoch(data: dict | list) -> int | None:
    """Return epoch number (0-indexed) with lowest val_loss; falls back to train_loss."""
    if isinstance(data, list):
        epochs = [e for e in data if not e.get("partial")]
    elif isinstance(data, dict):
        epochs = [e for e in data.get("epochs", []) if not e.get("partial")]
        if not epochs:
            epochs = data.get("val", []) or data.get("train", [])
    else:
        return None

    for loss_key in ("val_loss", "train_loss"):
        candidates = [
            (e.get("epoch", i), e[loss_key])
            for i, e in enumerate(epochs)
            if loss_key in e and e[loss_key] is not None
        ]
        if candidates:
            return min(candidates, key=lambda x: x[1])[0]

    return None


def _fmt_timestamp(ts: str) -> str:
    """Format ISO-8601 timestamp to local human-readable string."""
    from datetime import datetime

    try:
        dt = datetime.fromisoformat(ts.replace("Z", "+00:00"))
        return dt.astimezone().strftime("%Y-%m-%d %H:%M")
    except Exception:
        return ts

def _section_label(text: str) -> QLabel:
    lbl = QLabel(text.upper())
    f = QFont()
    f.setPointSize(9)
    f.setBold(True)
    lbl.setFont(f)
    lbl.setStyleSheet(
        f"color: {theme.MAUVE}; letter-spacing: 1px; background: transparent;"
    )
    return lbl


def _epoch_stats_row(completed: int, total: int, best: int | None) -> QWidget:
    row = QWidget()
    row.setStyleSheet("background: transparent;")
    hl = QHBoxLayout(row)
    hl.setContentsMargins(0, 0, 0, 0)
    hl.setSpacing(8)

    hl.addWidget(_big_card(str(total) if total else str(completed), "Total Epochs", theme.BLUE))
    completed_color = theme.GREEN if (total == 0 or completed >= total) else theme.PEACH
    hl.addWidget(_big_card(str(completed), "Completed", completed_color))
    # Display best epoch as 1-indexed (engineers think epoch 1, not epoch 0)
    best_text = str(best + 1) if best is not None else "—"
    hl.addWidget(_big_card(best_text, "Best Epoch", theme.MAUVE))
    return row


def _big_card(value: str, label: str, bg_color: str) -> QFrame:
    card = QFrame()
    card.setStyleSheet(f"QFrame {{ background-color: {bg_color}; border-radius: 8px; }}")
    card.setSizePolicy(QSizePolicy.Policy.Expanding, QSizePolicy.Policy.Fixed)
    vl = QVBoxLayout(card)
    vl.setContentsMargins(16, 14, 16, 14)
    vl.setSpacing(4)

    val_lbl = QLabel(value)
    vf = QFont()
    vf.setPointSize(20)
    vf.setBold(True)
    val_lbl.setFont(vf)
    val_lbl.setAlignment(Qt.AlignmentFlag.AlignCenter)
    val_lbl.setStyleSheet(f"color: {theme.BASE}; background: transparent;")
    vl.addWidget(val_lbl)

    key_lbl = QLabel(label)
    key_lbl.setAlignment(Qt.AlignmentFlag.AlignCenter)
    key_lbl.setStyleSheet(f"color: {theme.BASE}; font-size: 10pt; background: transparent;")
    vl.addWidget(key_lbl)
    return card


def _metric_grid(metrics: dict) -> QWidget:
    container = QWidget()
    container.setStyleSheet("background: transparent;")
    grid = QGridLayout(container)
    grid.setContentsMargins(0, 0, 0, 0)
    grid.setSpacing(6)

    items = sorted(
        ((k, v) for k, v in metrics.items() if k not in ("epoch", "phase", "global_step")),
        key=lambda x: x[0],
    )
    for idx, (key, value) in enumerate(items):
        if isinstance(value, float):
            text = f"{value:.4f}"
            color = _metric_color(key, value)
        else:
            text = str(value)
            color = theme.TEXT
        grid.addWidget(_metric_card(key, text, color), idx // 3, idx % 3)

    for c in range(3):
        grid.setColumnStretch(c, 1)
    return container


def _metric_card(key: str, value: str, color: str) -> QFrame:
    card = QFrame()
    card.setStyleSheet(f"QFrame {{ background-color: {theme.SURFACE0}; border-radius: 8px; }}")
    card.setSizePolicy(QSizePolicy.Policy.Expanding, QSizePolicy.Policy.Fixed)
    vl = QVBoxLayout(card)
    vl.setContentsMargins(10, 8, 10, 8)
    vl.setSpacing(2)

    name = key.split("_", 1)[-1].replace("_", " ").title()
    key_lbl = QLabel(name)
    key_lbl.setStyleSheet(
        f"color: {theme.SUBTEXT1}; font-size: 9pt; background: transparent;"
    )
    vl.addWidget(key_lbl)

    val_lbl = QLabel(value)
    vf = QFont()
    vf.setPointSize(13)
    vf.setBold(True)
    val_lbl.setFont(vf)
    val_lbl.setStyleSheet(f"color: {color}; background: transparent;")
    vl.addWidget(val_lbl)
    return card


def _metric_color(key: str, value: float) -> str:
    k = key.lower()
    if "loss" in k:
        return theme.RED if value > _LOSS_WARN else theme.GREEN
    if any(x in k for x in ("iou", "f1", "dice", "accuracy", "precision", "recall")):
        if value > _GOOD:
            return theme.GREEN
        if value > _WARN:
            return theme.YELLOW
        return theme.RED
    return theme.TEXT


def _warnings_widget(warnings: list) -> QFrame:
    frame = QFrame()
    frame.setStyleSheet(
        f"QFrame {{ background-color: {theme.SURFACE0}; border-radius: 8px; "
        f"border-left: 3px solid {theme.YELLOW}; }}"
    )
    vl = QVBoxLayout(frame)
    vl.setContentsMargins(14, 10, 14, 10)
    vl.setSpacing(6)

    title = QLabel("Training Warnings")
    tf = QFont()
    tf.setBold(True)
    tf.setPointSize(11)
    title.setFont(tf)
    title.setStyleSheet(f"color: {theme.YELLOW}; background: transparent;")
    vl.addWidget(title)

    for w in warnings:
        msg = QLabel(w.get("message", str(w)))
        msg.setWordWrap(True)
        msg.setStyleSheet(f"color: {theme.TEXT}; background: transparent; padding: 4px;")
        vl.addWidget(msg)
    return frame

def _compute_stats(results: list) -> dict:
    total = len(results)
    confidences = [r.confidence for r in results if r.confidence is not None]
    avg_conf = sum(confidences) / len(confidences) if confidences else 0.0
    crack_count = sum(1 for r in results if r.has_crack)
    high = sum(1 for r in results if r.crack_severity == "Crack Detected")
    suspected = sum(1 for r in results if r.crack_severity == "Suspected Crack")
    no_crack = total - high - suspected
    threshold = results[0].threshold if results else 0.5

    # IoU — only present when ground truth masks were available
    iou_vals = [r.iou for r in results if r.iou is not None]
    mean_iou: float | None = sum(iou_vals) / len(iou_vals) if iou_vals else None

    # Crack area % — only for images where a crack was detected
    area_vals = [
        r.crack_area_pct
        for r in results
        if r.has_crack and r.crack_area_pct is not None
    ]
    mean_crack_area_pct: float | None = (
        sum(area_vals) / len(area_vals) if area_vals else None
    )

    return {
        "total": total,
        "confidences": confidences,
        "avg_confidence": avg_conf,
        "crack_count": crack_count,
        "high": high,
        "suspected": suspected,
        "no_crack": no_crack,
        "threshold": threshold,
        "mean_iou": mean_iou,
        "mean_crack_area_pct": mean_crack_area_pct,
    }

def _section_label(text: str) -> QLabel:
    lbl = QLabel(text.upper())
    f = QFont()
    f.setPointSize(9)
    f.setBold(True)
    lbl.setFont(f)

def _stats_row(stats: dict) -> QWidget:
    row = QWidget()
    row.setStyleSheet("background: transparent;")
    hl = QHBoxLayout(row)
    hl.setContentsMargins(0, 0, 0, 0)
    hl.setSpacing(8)

    hl.addWidget(_big_card(str(stats["total"]), "Total Images", theme.BLUE))

    avg = stats["avg_confidence"]
    conf_color = theme.GREEN if avg > _GOOD else theme.PEACH if avg > _WARN else theme.RED
    conf_text = f"{avg:.3f}" if avg > 0 else "N/A"
    hl.addWidget(_big_card(conf_text, "Avg Confidence", conf_color))

    # Threshold is configuration, not a result — use a muted surface card
    hl.addWidget(
        _big_card(
            f"{stats['threshold']:.2f}",
            "Threshold",
            theme.SURFACE0,
            text_color=theme.SUBTEXT1,
        )
    )

    if stats["mean_iou"] is not None:
        iou = stats["mean_iou"]
        iou_color = theme.GREEN if iou > _GOOD else theme.PEACH if iou > _WARN else theme.RED
        hl.addWidget(_big_card(f"{iou:.3f}", "Mean IoU", iou_color))

    if stats["mean_crack_area_pct"] is not None:
        area = stats["mean_crack_area_pct"]
        area_color = theme.RED if area > 20 else theme.PEACH
        hl.addWidget(_big_card(f"{area:.1f}%", "Avg Crack Area", area_color))

    return row


def _big_card(value: str, label: str, bg_color: str, *, text_color: str = theme.BASE) -> QFrame:
    card = QFrame()
    card.setStyleSheet(f"QFrame {{ background-color: {bg_color}; border-radius: 8px; }}")
    card.setSizePolicy(QSizePolicy.Policy.Expanding, QSizePolicy.Policy.Fixed)
    vl = QVBoxLayout(card)
    vl.setContentsMargins(16, 14, 16, 14)
    vl.setSpacing(4)

    val_lbl = QLabel(value)
    vf = QFont()
    vf.setPointSize(20)
    vf.setBold(True)
    val_lbl.setFont(vf)
    val_lbl.setAlignment(Qt.AlignmentFlag.AlignCenter)
    val_lbl.setStyleSheet(f"color: {text_color}; background: transparent;")
    vl.addWidget(val_lbl)

    key_lbl = QLabel(label)
    key_lbl.setAlignment(Qt.AlignmentFlag.AlignCenter)
    key_lbl.setStyleSheet(f"color: {text_color}; font-size: 10pt; background: transparent;")
    vl.addWidget(key_lbl)
    return card


def _distribution_widget(stats: dict) -> QWidget:
    total = stats["total"]
    has_severity = stats["high"] > 0 or stats["suspected"] > 0

    w = QWidget()
    w.setStyleSheet("background: transparent;")
    vl = QVBoxLayout(w)
    vl.setContentsMargins(0, 0, 0, 0)
    vl.setSpacing(6)

    if has_severity:
        vl.addWidget(_dist_row("Crack Detected", stats["high"], total, theme.RED))
        vl.addWidget(_dist_row("Suspected Crack", stats["suspected"], total, theme.YELLOW))
        vl.addWidget(_dist_row("No Cracks", stats["no_crack"], total, theme.GREEN))
    else:
        vl.addWidget(_dist_row("Cracks Detected", stats["crack_count"], total, theme.RED))
        vl.addWidget(_dist_row("No Cracks", total - stats["crack_count"], total, theme.GREEN))
    return w


def _dist_row(label: str, count: int, total: int, color: str) -> QWidget:
    row = QWidget()
    row.setStyleSheet("background: transparent;")
    hl = QHBoxLayout(row)
    hl.setContentsMargins(0, 2, 0, 2)
    hl.setSpacing(10)

    lbl = QLabel(label)
    lbl.setFixedWidth(140)
    lbl.setStyleSheet(f"color: {theme.TEXT}; background: transparent;")
    hl.addWidget(lbl)

    bar = QProgressBar()
    bar.setRange(0, max(total, 1))
    bar.setValue(count)
    bar.setTextVisible(False)
    bar.setFixedHeight(18)
    bar.setStyleSheet(
        f"QProgressBar {{ background: {theme.SURFACE1}; border-radius: 4px; }}"
        f"QProgressBar::chunk {{ background: {color}; border-radius: 4px; }}"
    )
    hl.addWidget(bar, 1)

    pct = count / total * 100 if total > 0 else 0
    count_lbl = QLabel(f"{count}  ({pct:.1f}%)")
    count_lbl.setFixedWidth(100)
    count_lbl.setStyleSheet(f"color: {color}; background: transparent;")
    hl.addWidget(count_lbl)
    return row


def _confidence_widget(stats: dict) -> QWidget:
    confs = stats["confidences"]
    total = len(confs)
    high = sum(1 for c in confs if c > _GOOD)
    med = sum(1 for c in confs if _WARN < c <= _GOOD)
    low = sum(1 for c in confs if c <= _WARN)

    w = QWidget()
    w.setStyleSheet("background: transparent;")
    vl = QVBoxLayout(w)
    vl.setContentsMargins(0, 0, 0, 0)
    vl.setSpacing(6)
    vl.addWidget(_dist_row("High  (>0.70)", high, total, theme.GREEN))
    vl.addWidget(_dist_row("Medium  (0.50–0.70)", med, total, theme.YELLOW))
    vl.addWidget(_dist_row("Low  (<0.50)", low, total, theme.RED))
    return w
