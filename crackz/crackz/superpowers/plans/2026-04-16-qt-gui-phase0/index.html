    Qt GUI Phase 0 Implementation Plan¶

    For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

    Goal: Build the PySide6 Qt shell (activity bar + side panel + main content + status bar + AI chat drawer) with Catppuccin Mocha theming, Explorer panel, Gallery+Detail image viewer, and Welcome view — leaving the existing CustomTkinter GUI untouched.

    Architecture: New Qt code lives entirely in src/crackz/gui/qt/ and src/crackz/gui/style/, exposed via a new crackz-gui-qt entry point. The CTk GUI continues to work unchanged. AIChatService and AsyncEventLoop are reused without modification. QSS tokens are injected at startup via string.Template from theme.py constants.

    Tech Stack: PySide6 (QWidget, QMainWindow, QSplitter, QStackedWidget, QTreeView, QGraphicsView, QPropertyAnimation, QStatusBar), string.Template for QSS, pytest + QT_QPA_PLATFORM=offscreen for headless widget tests.

"""Qt status bar for Crackz — project name · counts · version."""
from __future__ import annotations

from PySide6.QtCore import Qt
from PySide6.QtWidgets import QHBoxLayout, QLabel, QStatusBar, QWidget

from crackz.version import __version__


class CrackzStatusBar(QStatusBar):
    """24px status bar: green dot + project name | summary counts | version."""

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self.setSizeGripEnabled(False)

        # Left: project indicator
        self._project_container = QWidget()
        left_layout = QHBoxLayout(self._project_container)
        left_layout.setContentsMargins(8, 0, 0, 0)
        left_layout.setSpacing(4)

        self._dot = QLabel("●")
        self._dot.setStyleSheet("color: #a6e3a1;")  # GREEN — not via QSS so it stays always
        self._dot.setVisible(False)

        self.project_label = QLabel("")
        left_layout.addWidget(self._dot)
        left_layout.addWidget(self.project_label)
        self.addWidget(self._project_container)

        # Center: image counts
        self.counts_label = QLabel("")
        self.counts_label.setAlignment(Qt.AlignCenter)
        self.addPermanentWidget(self.counts_label)

        # Right: version
        self.version_label = QLabel(f"v{__version__}")
        self.version_label.setAlignment(Qt.AlignRight)
        self.addPermanentWidget(self.version_label)

    def set_project(self, name: str | None) -> None:
        """Update the project name display."""
        if name:
            self.project_label.setText(name)
            self._dot.setVisible(True)
        else:
            self.project_label.setText("")
            self._dot.setVisible(False)

    def set_counts(self, *, total: int, critical: int, medium: int) -> None:
        """Update the detection summary counts."""
        self.counts_label.setText(
            f"{total} images · {critical} critical · {medium} medium"
        )

    def clear_counts(self) -> None:
        """Clear the detection summary."""
        self.counts_label.setText("")

from __future__ import annotations

_ICONS_DIR = Path(__file__).parents[2] / "assets" / "icons"

_ACTIVITIES = [
    ("explorer", "Explorer"),
    ("detect", "Detect"),
    ("train", "Train"),
    ("active_learning", "Active Learning"),
    ("settings", "Settings"),

class ActivityBar(QWidget):
    """44px-wide vertical bar with icon buttons that switch the active panel.

    Signals
    -------
    activity_changed(int)
        Emitted with the new panel index when the active activity changes.
    """

    activity_changed = Signal(int)

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self.setObjectName("ActivityBar")
        self.setFixedWidth(44)

        layout = QVBoxLayout(self)
        layout.setContentsMargins(0, 8, 0, 8)
        layout.setSpacing(2)

        self.buttons: list[QToolButton] = []
        self.active_index: int = 0

        # Build buttons for Explorer/Detect/Train/ActiveLearning
        for idx, (icon_name, label) in enumerate(_ACTIVITIES[:-1]):
            btn = self._make_button(icon_name, label, idx)
            layout.addWidget(btn)
            self.buttons.append(btn)

        layout.addSpacerItem(QSpacerItem(0, 0, QSizePolicy.Minimum, QSizePolicy.Expanding))

        # Settings at the bottom
        idx = len(_ACTIVITIES) - 1
        icon_name, label = _ACTIVITIES[idx]
        btn = self._make_button(icon_name, label, idx)
        layout.addWidget(btn)
        self.buttons.append(btn)

        # Activate the first button
        self._apply_active(0)

    def _make_button(self, icon_name: str, tooltip: str, index: int) -> QToolButton:
        btn = QToolButton()
        btn.setToolTip(tooltip)
        btn.setIconSize(QSize(20, 20))
        btn.setFixedSize(QSize(44, 44))
        btn.setProperty("active", "false")

        icon_path = _ICONS_DIR / f"{icon_name}.svg"
        if icon_path.exists():
            btn.setIcon(QIcon(str(icon_path)))
        else:
            # Fallback: use first letter as text
            btn.setText(tooltip[0])

        btn.clicked.connect(lambda checked=False, i=index: self._on_click(i))
        return btn

    def _on_click(self, index: int) -> None:
        if index == self.active_index:
            return
        self.set_active(index)
        self.activity_changed.emit(index)

    def set_active(self, index: int) -> None:
        """Set the active activity without emitting a signal."""
        for i, btn in enumerate(self.buttons):
            btn.setProperty("active", "true" if i == index else "false")
            btn.style().unpolish(btn)
            btn.style().polish(btn)
        self.active_index = index

"""AI Chat Drawer — slides in from the right edge of the main content area.

Wraps the existing AIChatService without modifying it.
Animation: QPropertyAnimation on maximumWidth (0 ↔ 280px), 150ms ease-in-out.
"""
from __future__ import annotations

from PySide6.QtCore import (
    QEasingCurve,
    QPropertyAnimation,
    Qt,
    QTimer,
)
from PySide6.QtWidgets import (
    QHBoxLayout,
    QLabel,
    QLineEdit,
    QPushButton,
    QScrollArea,
    QSizePolicy,
    QToolButton,
    QVBoxLayout,
    QWidget,
)

_DRAWER_WIDTH = 280


class AIChatDrawer(QWidget):
    """Slide-in AI chat panel.

    Hidden by default (maximumWidth=0). Call toggle() or set the
    MainWindow shortcut (~  /  Cmd+/) to show/hide.

    The drawer creates its own AsyncEventLoop and AIChatService lazily on
    first show, so it never blocks startup.
    """

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self.setObjectName("AIChatDrawer")
        self.setMaximumWidth(0)
        self.setSizePolicy(QSizePolicy.Fixed, QSizePolicy.Expanding)
        self._is_open = False
        self._bubble_widgets: list[QWidget] = []
        self._service = None  # lazy init

        # ── Animation ─────────────────────────────────────────────
        self._anim = QPropertyAnimation(self, b"maximumWidth")
        self._anim.setDuration(150)
        self._anim.setEasingCurve(QEasingCurve.InOutQuad)

        # ── Layout ────────────────────────────────────────────────
        root = QVBoxLayout(self)
        root.setContentsMargins(0, 0, 0, 0)
        root.setSpacing(0)

        # Header
        header = QWidget()
        header.setObjectName("DrawerHeader")
        header_layout = QHBoxLayout(header)
        header_layout.setContentsMargins(12, 6, 8, 6)

        title = QLabel("AI ASSISTANT")
        title.setObjectName("SectionLabel")
        header_layout.addWidget(title)
        header_layout.addStretch()

        close_btn = QToolButton()
        close_btn.setText("×")
        close_btn.setToolTip("~ to close")
        close_btn.clicked.connect(self.toggle)
        header_layout.addWidget(close_btn)
        root.addWidget(header)

        # Chat scroll area
        self._scroll = QScrollArea()
        self._scroll.setWidgetResizable(True)
        self._scroll.setFrameShape(QScrollArea.NoFrame)

        self._chat_container = QWidget()
        self._chat_layout = QVBoxLayout(self._chat_container)
        self._chat_layout.setAlignment(Qt.AlignTop)
        self._chat_layout.setSpacing(8)
        self._chat_layout.setContentsMargins(8, 8, 8, 8)
        self._chat_layout.addStretch()
        self._scroll.setWidget(self._chat_container)
        root.addWidget(self._scroll, stretch=1)

        # Input row
        input_row = QWidget()
        input_layout = QHBoxLayout(input_row)
        input_layout.setContentsMargins(8, 6, 8, 8)
        input_layout.setSpacing(4)

        self._input = QLineEdit()
        self._input.setPlaceholderText("Ask about this image…")
        self._input.returnPressed.connect(self._on_return_pressed)
        input_layout.addWidget(self._input)

        send_btn = QPushButton("↵")
        send_btn.setFixedWidth(32)
        send_btn.clicked.connect(self._on_return_pressed)
        input_layout.addWidget(send_btn)
        root.addWidget(input_row)

    # ── Public API ────────────────────────────────────────────────

    def toggle(self) -> None:
        """Slide the drawer open or closed."""
        if self._is_open:
            self._anim.setStartValue(self.maximumWidth())
            self._anim.setEndValue(0)
            self._anim.start()
            self._is_open = False
        else:
            self._ensure_service()
            self._anim.setStartValue(self.maximumWidth())
            self._anim.setEndValue(_DRAWER_WIDTH)
            self._anim.start()
            self._is_open = True

    # ── Internal helpers ──────────────────────────────────────────

    def _ensure_service(self) -> None:
        """Lazily initialise AIChatService on first open."""
        if self._service is not None:
            return
        try:
            from crackz.gui.async_helpers import AsyncEventLoop
            from crackz.gui.services.ai_chat import AIChatService

            loop = AsyncEventLoop()
            self._service = AIChatService(async_loop=loop)
        except Exception:
            self._service = None  # remain unconfigured; chat will show error

    def _on_return_pressed(self) -> None:
        text = self._input.text().strip()
        if not text:
            return
        self._input.clear()
        self._on_send(text)

    def _on_send(self, message: str) -> None:
        self._add_bubble(message, is_user=True)
        self._send_to_ai(message)

    def _send_to_ai(self, message: str) -> None:
        if self._service is None:
            self._add_bubble("AI service not configured. Set your API key in settings.", is_user=False)
            return

        def callback(response: str, *, is_error: bool) -> None:
            # Schedule on main thread via QTimer.singleShot
            QTimer.singleShot(0, lambda: self._add_bubble(response, is_user=False))

        self._service.send_message(message, callback)

    def _add_bubble(self, text: str, *, is_user: bool) -> None:
        """Add a chat bubble to the scroll area."""
        bubble = QLabel(text)
        bubble.setWordWrap(True)
        bubble.setObjectName("UserBubble" if is_user else "AssistantBubble")
        bubble.setAlignment(Qt.AlignRight if is_user else Qt.AlignLeft)
        bubble.setStyleSheet(
            "background-color: #313244; border-radius: 8px; padding: 6px 10px;"
            if is_user
            else "background-color: rgba(203,166,247,0.13); border-radius: 8px; padding: 6px 10px;"
        )

        row = QWidget()
        row_layout = QHBoxLayout(row)
        row_layout.setContentsMargins(0, 0, 0, 0)
        if is_user:
            row_layout.addStretch()
            row_layout.addWidget(bubble)
        else:
            row_layout.addWidget(bubble)
            row_layout.addStretch()

        # Insert before the trailing stretch
        count = self._chat_layout.count()
        self._chat_layout.insertWidget(count - 1, row)
        self._bubble_widgets.append(row)

        # Scroll to bottom
        QTimer.singleShot(0, lambda: self._scroll.verticalScrollBar().setValue(
            self._scroll.verticalScrollBar().maximum()
        ))

from string import Template

from crackz.gui.style import theme


def create_app() -> QApplication:
    """Create and configure the QApplication with the Crackz stylesheet."""
    app = QApplication.instance() or QApplication(sys.argv)

    qss_path = Path(__file__).parents[2] / "style" / "crackz.qss"

    return app

    from crackz.gui.qt.main_window import MainWindow

    app = create_app()
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
