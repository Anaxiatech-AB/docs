    PySide6 Migration — Phase 0 Spike Implementation Plan¶

    For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

    Goal: Prove out the PySide6 migration approach by building a runnable crackz-qt entry point: dark QSS shell, QMainWindow, async bridge, and one real ported panel (ProjectSummaryView).

    Architecture: A parallel src/crackz/qt/ package lives alongside the existing src/crackz/gui/ (which keeps shipping CTk throughout Phase 0). The Qt app is launched via a new crackz-qt CLI script. No CTk code is modified. At the end of Phase 0 you can launch both and compare side by side.

    Tech Stack: PySide6 6.x, Python 3.13, uv, pytest (headless with QT_QPA_PLATFORM=offscreen), PyInstaller for packaging smoke test.

    Branch: feat/pyside6-migration

    src/crackz/qt/async_bridge.py: 

    """Qt-compatible async event loop and thread-safe GUI callbacks.

Replaces `src/crackz/gui/async_helpers.py` for the PySide6 migration.

Key difference from the CTk version:
  - `create_threadsafe_gui_callback` no longer needs a root widget reference.
  - Qt signals emitted from non-GUI threads are automatically queued to the
    GUI thread by Qt's event system — no tk.after() needed.
  - Callers must store the returned `emitter` object to prevent garbage
    collection (the signal connection is held by the emitter).
"""

from __future__ import annotations

import asyncio
import threading
from typing import TYPE_CHECKING, Any

from PySide6.QtCore import QObject, Signal, Slot

from crackz.log import logger

if TYPE_CHECKING:
    from collections.abc import Callable

    from crackz.ai.core.async_callbacks import CallbackDispatcher


# ---------------------------------------------------------------------------
# Signal emitter helpers
# ---------------------------------------------------------------------------


class _ProgressEmitter(QObject):
    """Internal QObject that owns the progress signal."""

    # No type annotation — PySide6 metaclass requires bare class-var assignment
    progress_received = Signal(float, object)


class _MetricsEmitter(QObject):
    """Internal QObject that owns the metrics signal."""

    metrics_received = Signal(dict)


# ---------------------------------------------------------------------------
# Async event loop (identical API to gui/async_helpers.AsyncEventLoop)
# ---------------------------------------------------------------------------


class AsyncEventLoop:
    """Manages an asyncio event loop in a daemon thread for Qt GUI integration.

    API-compatible drop-in for the Tkinter-based version in gui/async_helpers.py.
    """

    def __init__(self) -> None:
        self._loop: asyncio.AbstractEventLoop | None = None
        self._thread: threading.Thread | None = None
        self._ready = threading.Event()
        self._running = False

    def start(self) -> None:
        """Start the event loop in a background daemon thread."""
        if self._running:
            logger.warning("AsyncEventLoop already running")
            return
        self._running = True
        self._ready.clear()

        def _run() -> None:
            try:
                self._loop = asyncio.new_event_loop()
                asyncio.set_event_loop(self._loop)
                self._ready.set()
                self._loop.run_forever()
            except Exception as e:
                logger.error("Error in async event loop: {}", e)
            finally:
                if self._loop:
                    self._loop.close()

        self._thread = threading.Thread(target=_run, daemon=True, name="QtAsyncEventLoop")
        self._thread.start()

        if not self._ready.wait(timeout=5.0):
            self._running = False
            raise RuntimeError("Event loop failed to start within 5 seconds")

    def stop(self) -> None:
        """Stop the event loop and join the thread."""
        if not self._running:
            return
        self._running = False
        if self._loop:
            self._loop.call_soon_threadsafe(self._loop.stop)
        if self._thread:
            self._thread.join(timeout=2.0)
            if self._thread.is_alive():
                logger.warning("Async event loop thread did not terminate gracefully")

    def run_coroutine(self, coro: Any) -> asyncio.Future:  # noqa: ANN401
        """Schedule a coroutine and return a concurrent.futures.Future."""
        if not self._loop:
            raise RuntimeError("Event loop not started")
        return asyncio.run_coroutine_threadsafe(coro, self._loop)

    def create_task(self, coro: Any) -> None:  # noqa: ANN401
        """Fire-and-forget: schedule a coroutine as a task."""
        if not self._loop:
            raise RuntimeError("Event loop not started")

        def _create() -> None:
            task = asyncio.ensure_future(coro, loop=self._loop)
            task.add_done_callback(lambda _: None)

        self._loop.call_soon_threadsafe(_create)


# ---------------------------------------------------------------------------
# Thread-safe callbacks via Qt signals
# ---------------------------------------------------------------------------


def create_threadsafe_gui_callback(
    callback: Callable[[float, str | None], None],
) -> tuple[Callable[[float, str | None], None], _ProgressEmitter]:
    """Create a thread-safe progress callback using Qt signals.

    Qt signals emitted from any thread are automatically queued to the GUI
    thread, replacing the tk.after(0, ...) pattern.

    Args:
        callback: The GUI-thread function to invoke on progress updates.

    Returns:
        (thread_safe_fn, emitter) — **store the emitter** to prevent GC.

    Example::

        safe_cb, _emitter = create_threadsafe_gui_callback(self.on_progress)
        self._emitters.append(_emitter)  # keep alive
    """
    emitter = _ProgressEmitter()
    emitter.progress_received.connect(callback)

    def safe_callback(progress: float, message: str | None = None) -> None:
        emitter.progress_received.emit(progress, message)

    return safe_callback, emitter


def create_threadsafe_metrics_callback(
    callback: Callable[[dict[str, float]], None],
) -> tuple[Callable[[dict[str, float]], None], _MetricsEmitter]:
    """Create a thread-safe metrics callback using Qt signals.

    Args:
        callback: The GUI-thread function to invoke on metrics updates.

    Returns:
        (thread_safe_fn, emitter) — **store the emitter** to prevent GC.
    """
    emitter = _MetricsEmitter()
    emitter.metrics_received.connect(callback)

    def safe_callback(metrics: dict[str, float]) -> None:
        emitter.metrics_received.emit(metrics.copy())

    return safe_callback, emitter


# ---------------------------------------------------------------------------
# High-level consumer helpers (mirrors gui/async_helpers.start_gui_callback_consumers)
# ---------------------------------------------------------------------------


def start_gui_callback_consumers(
    dispatcher: CallbackDispatcher,
    progress_callback: Callable[[float, str | None], None],
    metrics_callback: Callable[[dict[str, float]], None] | None = None,
) -> tuple[AsyncEventLoop, list[asyncio.Task], list[QObject]]:
    """Start async callback consumers for Qt GUI integration.

    Args:
        dispatcher: CallbackDispatcher from the AI pipeline.
        progress_callback: GUI-thread function for progress updates.
        metrics_callback: Optional GUI-thread function for metrics updates.

    Returns:
        (loop, tasks, emitters) — store **emitters** in your widget to
        prevent garbage collection of the signal connections.
    """
    from crackz.ai.core.async_callbacks import (
        create_async_metrics_consumer,
        create_async_progress_consumer,
    )

    emitters: list[QObject] = []

    safe_progress, prog_emitter = create_threadsafe_gui_callback(progress_callback)
    emitters.append(prog_emitter)

    safe_metrics = None
    if metrics_callback:
        safe_metrics_fn, met_emitter = create_threadsafe_metrics_callback(metrics_callback)
        safe_metrics = safe_metrics_fn
        emitters.append(met_emitter)

    loop = AsyncEventLoop()
    loop.start()
    tasks: list[asyncio.Task] = []

    progress_consumer = create_async_progress_consumer(dispatcher, safe_progress)

    async def _run_progress() -> None:
        try:
            await progress_consumer()
        except Exception as e:
            logger.error("Progress consumer error: {}", e)

    loop.create_task(_run_progress())

    if safe_metrics:
        metrics_consumer = create_async_metrics_consumer(dispatcher, safe_metrics)

        async def _run_metrics() -> None:
            try:
                await metrics_consumer()
            except Exception as e:
                logger.error("Metrics consumer error: {}", e)

        loop.create_task(_run_metrics())

    return loop, tasks, emitters


def stop_gui_callback_consumers(
    dispatcher: CallbackDispatcher,
    loop: AsyncEventLoop,
) -> None:
    """Stop async callback consumers and clean up.

    Args:
        dispatcher: CallbackDispatcher to close.
        loop: AsyncEventLoop to stop.
    """
    import time

    async def _close() -> None:
        await dispatcher.close_all()

    if loop._loop:
        future = loop.run_coroutine(_close())
        try:
            future.result(timeout=2.0)
        except Exception as e:
            logger.warning("Error closing dispatcher: {}", e)
        time.sleep(0.1)

    loop.stop()

    """Runtime application state for the Qt GUI.

Equivalent to src/crackz/gui/gui_state.py but uses QPixmap instead of
CTkImage so there are no Tkinter imports in the Qt package.
"""

from __future__ import annotations

from collections import OrderedDict
from dataclasses import dataclass, field
from typing import TYPE_CHECKING

from diskcache import Cache
from PIL import Image

from crackz.log import logger

if TYPE_CHECKING:
    from pathlib import Path

    from PySide6.QtGui import QPixmap

    from crackz import Project
    from crackz.project import SizeType


@dataclass
class QtAppState:
    """Runtime state for the Qt GUI.

    Holds the active project and a two-tier thumbnail cache
    (in-memory QPixmap LRU + persistent diskcache).
    """

    project: Project | None = None
    project_path: Path | None = None

    # Dashboard server state
    dashboard_running: bool = field(default=False, init=False)
    dashboard_port: int | None = field(default=None, init=False)
    dashboard_url: str | None = field(default=None, init=False)

    thumbnail_cache: Cache | None = field(default=None, init=False)
    _pixmap_cache: OrderedDict[tuple, QPixmap] = field(
        default_factory=OrderedDict, init=False
    )
    _max_memory_cache_size: int = field(default=500, init=False)

    def __post_init__(self) -> None:
        if self.project_path:
            self._init_disk_cache(self.project_path)

    def _init_disk_cache(self, project_path: Path) -> None:
        try:
            cache_dir = project_path / ".thumbnails"
            cache_dir.mkdir(exist_ok=True)
            self.thumbnail_cache = Cache(
                str(cache_dir),
                size_limit=100 * 1024 * 1024,
                eviction_policy="least-recently-used",
            )
        except Exception as e:
            logger.warning("Failed to initialize disk cache: {}", e)
            self.thumbnail_cache = None

    def get_thumbnail(self, path: Path, size: SizeType = (150, 150)) -> QPixmap | None:
        """Return a cached QPixmap thumbnail, creating it if necessary.

        Two-tier cache: in-memory QPixmap LRU (fast) → disk PIL cache (warm)
        → create from original (slow).
        """
        from PySide6.QtGui import QPixmap as _QPixmap

        try:
            mtime = path.stat().st_mtime
        except FileNotFoundError:
            return None

        key = (str(path), size, mtime)
        cache_key = f"{path.stem}_{size[0]}x{size[1]}_{int(mtime)}"

        # 1. Memory cache
        if key in self._pixmap_cache:
            self._pixmap_cache.move_to_end(key)
            return self._pixmap_cache[key]

        # 2. Disk cache
        pil_img: Image.Image | None = None
        if self.thumbnail_cache is not None:
            try:
                pil_img = self.thumbnail_cache.get(cache_key)
            except Exception:
                pil_img = None

        # 3. Create from original
        if pil_img is None:
            try:
                img = Image.open(path)
                if img.mode not in ("RGB", "L"):
                    img = img.convert("RGB")
                pil_img = img.resize(size, Image.Resampling.LANCZOS)
                if self.thumbnail_cache is not None:
                    try:
                        self.thumbnail_cache.set(cache_key, pil_img)
                    except Exception:
                        pass
            except Exception as e:
                logger.warning("Failed to create thumbnail for {}: {}", path, e)
                return None

        # Convert PIL → QPixmap
        try:
            import io

            from PySide6.QtCore import Qt

            buf = io.BytesIO()
            pil_img.save(buf, format="PNG")
            pixmap = _QPixmap()
            pixmap.loadFromData(buf.getvalue())
            if pixmap.isNull():
                return None
            pixmap = pixmap.scaled(
                size[0],
                size[1],
                Qt.AspectRatioMode.KeepAspectRatio,
                Qt.TransformationMode.SmoothTransformation,
            )
        except Exception as e:
            logger.warning("Failed to convert thumbnail to QPixmap: {}", e)
            return None

        # Store in memory cache with LRU eviction
        self._pixmap_cache[key] = pixmap
        if len(self._pixmap_cache) > self._max_memory_cache_size:
            self._pixmap_cache.popitem(last=False)

        return pixmap

    def clear_thumbnail_cache(self) -> None:
        """Clear both in-memory and disk thumbnail caches."""
        self._pixmap_cache.clear()
        if self.thumbnail_cache is not None:
            try:
                self.thumbnail_cache.clear()
            except Exception as e:
                logger.warning("Failed to clear disk cache: {}", e)

from __future__ import annotations

from abc import abstractmethod
from typing import TYPE_CHECKING

from PySide6.QtWidgets import QWidget

if TYPE_CHECKING:
    from crackz.gui.qt.app_state import QtAppState
    from crackz.gui.qt.main_window import MainWindow


class BaseView(QWidget):
    """Abstract base for all right-pane content views.

    Subclasses implement ``_build_ui()`` and optionally ``on_show()``,
    ``on_hide()``, and ``refresh()``.
    """

    def __init__(
        self,
        app_state: QtAppState,
        window: MainWindow,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)
        self.app_state = app_state
        self.window = window
        self._build_ui()

    @abstractmethod
    def _build_ui(self) -> None:
        """Build the view's widgets. Called once during ``__init__``."""

    def on_show(self) -> None:
        """Called when the view becomes the active right-pane content."""

    def on_hide(self) -> None:
        """Called just before the view is replaced or removed."""

    def refresh(self) -> None:
        """Reload data and rebuild the view. Override if needed."""
